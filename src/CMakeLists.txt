cmake_minimum_required(VERSION 3.20)
project(WireViewPro2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Afterburner requires a 32-bit plugin. Configure with -A Win32.")
  endif()
  set(AFTERBURNER_SDK "C:/Program Files (x86)/MSI Afterburner/SDK" CACHE PATH "Installed MSI Afterburner SDK")
  if(NOT EXISTS "${AFTERBURNER_SDK}/Include/MSIAfterburnerMonitoringSourceDesc.h")
    message(FATAL_ERROR "MSI Afterburner SDK not found. Set AFTERBURNER_SDK to its SDK directory.")
  endif()
  add_library(WireViewPro2 SHARED plugin.cpp monitor.cpp settings.cpp WireViewPro2.def)
  target_include_directories(WireViewPro2 PRIVATE "${AFTERBURNER_SDK}/Include")
  target_compile_definitions(WireViewPro2 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(WireViewPro2 PRIVATE setupapi advapi32 user32 gdi32)
  set_property(TARGET WireViewPro2 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# Local tests are optional and are not included in the public repository.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
  include(CTest)
  if(BUILD_TESTING)
    add_subdirectory(tests)
  endif()
endif()
